### Player State Machine

```javascript
class Player {
  constructor() {
    // Position & Physics
    this.x = 100;
    this.y = 450;
    this.vx = 0;  // velocity X
    this.vy = 0;  // velocity Y
    this.ax = 0;  // acceleration X
    this.ay = PHYSICS.GRAVITY_BASE;
    
    // Form System
    this.form = 'circle'; // 'circle' | 'triangle' | 'square'
    this.size = 20; // hitbox radius/half-width
    
    // Penalty Trackers (reset per level)
    this.gravityMultiplier = 1.0;
    this.speedDecay = 1.0;
    this.idleTime = 0;
    this.jumpCount = 0;
    
    // State
    this.grounded = false;
    this.alive = true;
  }
  
  // Form-specific modifiers
  getFormStats() {
    switch(this.form) {
      case 'circle':
        return { speed: 1.5, jump: 1.0, penalty: 'shrink' };
      case 'triangle':
        return { speed: 1.0, jump: 1.4, penalty: 'gravity' };
      case 'square':
        return { speed: 0.8, jump: 1.0, penalty: 'deceleration' };
    }
  }
}
```

### Collision Detection (AABB + SAT)

```javascript
function checkCollision(player, obstacle) {
  // Axis-Aligned Bounding Box (rectangular hitboxes)
  const px = player.x, py = player.y, ps = player.size;
  const ox = obstacle.x, oy = obstacle.y;
  const ow = obstacle.width, oh = obstacle.height;
  
  // Rectangular overlap test
  return (
    px + ps > ox &&
    px - ps < ox + ow &&
    py + ps > oy &&
    py - ps < oy + oh
  );
}

function resolveCollision(player, platform) {
  // Determine collision side
  const overlapX = Math.min(
    player.x + player.size - platform.x,
    platform.x + platform.width - (player.x - player.size)
  );
  const overlapY = Math.min(
    player.y + player.size - platform.y,
    platform.y + platform.height - (player.y - player.size)
  );
  
  // Push out of collision on smallest overlap axis
  if (overlapX < overlapY) {
    player.x += player.vx > 0 ? -overlapX : overlapX;
    player.vx = 0;
  } else {
    player.y += player.vy > 0 ? -overlapY : overlapY;
    player.vy = 0;
    player.grounded = (player.vy > 0); // landed on top
  }
}
```

## 🎨 RENDERING SYSTEM

### Draw Loop (Optimized)
```javascript
function render(ctx, camera) {
  // Clear with fade trail (motion blur effect)
  ctx.fillStyle = 'rgba(0, 0, 0, 0.1)';
  ctx.fillRect(0, 0, 800, 600);
  
  // Apply camera transform
  ctx.save();
  ctx.translate(-camera.x, -camera.y);
  
  // Draw platforms (white glow)
  ctx.shadowBlur = 15;
  ctx.shadowColor = '#fff';
  ctx.fillStyle = '#fff';
  
  platforms.forEach(p => {
    ctx.fillRect(p.x, p.y, p.width, p.height);
  });
  
  // Draw player (form-based shape)
  drawPlayer(ctx, player);
  
  ctx.restore();
}

function drawPlayer(ctx, p) {
  ctx.save();
  ctx.translate(p.x, p.y);
  
  // Rotation based on velocity
  const angle = Math.atan2(p.vy, p.vx);
  ctx.rotate(angle);
  
  ctx.fillStyle = '#fff';
  ctx.shadowBlur = 20;
  ctx.shadowColor = '#0ff';
  
  switch(p.form) {
    case 'circle':
      ctx.beginPath();
      ctx.arc(0, 0, p.size, 0, Math.PI * 2);
      ctx.fill();
      break;
    case 'triangle':
      ctx.beginPath();
      ctx.moveTo(0, -p.size);
      ctx.lineTo(p.size, p.size);
      ctx.lineTo(-p.size, p.size);
      ctx.closePath();
      ctx.fill();
      break;
    case 'square':
      ctx.fillRect(-p.size, -p.size, p.size * 2, p.size * 2);
      break;
  }
  
  ctx.restore();
}
```

## 🎯 ANIMATION SYSTEMS

### Smooth Interpolation (Easing)
```javascript
function easeOutCubic(t) {
  return 1 - Math.pow(1 - t, 3);
}

function animateDoorOpen(door, duration = 1000) {
  const start = Date.now();
  const startY = door.y;
  const targetY = door.y - 60;
  
  function step() {
    const elapsed = Date.now() - start;
    const progress = Math.min(elapsed / duration, 1);
    door.y = startY + (targetY - startY) * easeOutCubic(progress);
    
    if (progress < 1) requestAnimationFrame(step);
  }
  step();
}
```

## 🔧 INPUT HANDLING

```javascript
const keys = {};
document.addEventListener('keydown', e => keys[e.key] = true);
document.addEventListener('keyup', e => keys[e.key] = false);

function handleInput(player) {
  const stats = player.getFormStats();
  
  // Horizontal movement
  if (keys['ArrowLeft'] || keys['a']) {
    player.ax = -PHYSICS.MOVE_ACCELERATION * stats.speed;
    player.idleTime = 0;
  } else if (keys['ArrowRight'] || keys['d']) {
    player.ax = PHYSICS.MOVE_ACCELERATION * stats.speed;
    player.idleTime = 0;
  } else {
    player.ax = 0;
    player.idleTime++;
  }
  
  // Jump (only when grounded)
  if ((keys['ArrowUp'] || keys['w'] || keys[' ']) && player.grounded) {
    player.vy = PHYSICS.JUMP_IMPULSE * stats.jump;
    player.grounded = false;
    
    // Apply form-specific penalty
    if (player.form === 'triangle') {
      player.jumpCount++;
      player.gravityMultiplier += 0.05;
    }
  }
  
  // Form switch (1, 2, 3 keys)
  if (keys['1']) player.form = 'circle';
  if (keys['2']) player.form = 'triangle';
  if (keys['3']) player.form = 'square';
}
```

const gameState = {

          this.level++;
          this.transition('PLAYING');
        } else {

        }
        break;

  }
};
